if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The supplied MW2 executable is AMD64. Build with -A x64.")
  endif()
  add_library(d3d9 SHARED src/proxy.cpp src/mp_feedback.cpp src/overlay.cpp src/engine.cpp src/expanded_capacity.cpp src/ai.cpp src/memory.cpp src/zombies.cpp src/settings.cpp src/log.cpp src/static_watch.cpp src/script_vm.cpp src/bo1_assets.cpp src/bo1_models.cpp src/hang_watch.cpp third_party/miniz/miniz_tinfl.c src/d3d9.def)
  target_include_directories(d3d9 PRIVATE include third_party/miniz)
  target_compile_definitions(d3d9 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_link_libraries(d3d9 PRIVATE user32 gdi32 bcrypt dbghelp winmm advapi32)
  set_target_properties(d3d9 PROPERTIES PREFIX "" OUTPUT_NAME "d3d9")
  if(MSVC)
    set_property(TARGET d3d9 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(d3d9 PRIVATE /W4 /EHsc /permissive-)
  else()
    target_compile_options(d3d9 PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(d3d9 PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--dynamicbase,--nxcompat)
  endif()

  # Separate startup control: Win32 C only, no CRT, constructors, TLS, or hooks.
  add_library(d3d9_startup SHARED src/startup_proxy.c src/d3d9.def)
  add_executable(LoaderProbe tools/loader_probe.c)
  target_include_directories(d3d9_startup PRIVATE include)
  target_include_directories(LoaderProbe PRIVATE include)
  foreach(target d3d9_startup LoaderProbe)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /GS- /Oi-)
      target_link_options(${target} PRIVATE /NODEFAULTLIB /DYNAMICBASE /NXCOMPAT)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffreestanding -fno-builtin -fno-stack-protector)
      target_link_options(${target} PRIVATE -nostdlib -Wl,--dynamicbase,--nxcompat)
    endif()
    target_link_libraries(${target} PRIVATE kernel32)
  endforeach()
  target_link_libraries(LoaderProbe PRIVATE shell32)
  set_target_properties(d3d9_startup PROPERTIES PREFIX "" OUTPUT_NAME "d3d9"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/startup"
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/startup")
  if(MSVC)
    target_link_options(d3d9_startup PRIVATE /ENTRY:DllMain)
    target_link_options(LoaderProbe PRIVATE /ENTRY:ProbeEntry /SUBSYSTEM:CONSOLE)
  else()
    target_link_options(d3d9_startup PRIVATE -Wl,--entry,DllMain)
    target_link_options(LoaderProbe PRIVATE -Wl,--entry,ProbeEntry -Wl,--subsystem,console)
  endif()
endif()
