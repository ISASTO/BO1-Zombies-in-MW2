cmake_minimum_required(VERSION 3.20)
project(BO1MW2 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_include_directories(core_tests PRIVATE include)
  add_test(NAME core_tests COMMAND core_tests)
endif()
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The supplied MW2 executable is AMD64. Build with -A x64.")
  endif()
  add_library(d3d9 SHARED src/proxy.cpp src/overlay.cpp src/engine.cpp src/log.cpp src/d3d9.def)
  target_include_directories(d3d9 PRIVATE include)
  target_compile_definitions(d3d9 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_link_libraries(d3d9 PRIVATE user32 gdi32 bcrypt dbghelp)
  set_target_properties(d3d9 PROPERTIES PREFIX "" OUTPUT_NAME "d3d9")
  if(MSVC)
    set_property(TARGET d3d9 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(d3d9 PRIVATE /W4 /EHsc /permissive-)
  else()
    target_compile_options(d3d9 PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(d3d9 PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--dynamicbase,--nxcompat)
  endif()
endif()
